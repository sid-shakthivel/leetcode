// Title: Linked List Cycle
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/linked-list-cycle/

#         self.next = None

class Solution:
    def hasCycle(self, head: Optional[ListNode]) -> bool:
        slow = head
        fast = head

        while slow and fast and fast.next:
            slow = slow.next
            fast = fast.next.next

            if slow == fast:
                return True

        return False
