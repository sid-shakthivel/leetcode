// Title: Valid Palindrome
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/valid-palindrome/

class Solution:
    def isPalindrome(self, s: str) -> bool:
        s = re.sub('[^A-Za-z0-9]+', '', s).lower()

        for i in range(len(s)):
            if s[i] != s[len(s) - 1 - i]:
                return False

        return True
